# The landing page at / is public and wants to be found. The four
# post-checkout pages are not: they are reachable only through checkout, and
# each carries a noindex in its own meta tag and in the X-Robots-Tag header
# vercel.json sets on those four routes.
#
# Crawling is ALLOWED on all of it, on purpose. A crawler has to be able to
# fetch a page to see either of those noindex signals. Disallowing the funnel
# paths here would hide the noindex from the crawler while leaving the URLs
# themselves indexable from any external link, which is the opposite of what
# it looks like it does.
User-agent: *
Allow: /
